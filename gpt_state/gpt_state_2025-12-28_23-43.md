from fastapi.responses import JSONResponse
import json, os
from pathlib import Path

app = FastAPI()

@app.get("/metrics")
async def get_learning_metrics():
    metrics_path = Path(__file__).resolve().parent.parent / "state" / "learning_metrics.json"
    if not metrics_path.exists():
        return JSONResponse(content={"error": "learning_metrics.json not found"})
    with open(metrics_path, "r") as f:
        data = json.load(f)
    return JSONResponse(content=data)

from fastapi import FastAPI
from api.learning_routes import app as learning_app
from fastapi.middleware.cors import CORSMiddleware
from astra_dashboard.fetch_core.api_router import get_best

app = FastAPI(title="Astra Intelligence API", version="1.0")
app.mount("/learning", learning_app)

# Enable CORS
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"]
)

@app.get("/")
def root():
    return {"status": "Astra API online"}

@app.get("/learning/status")
def learning_status():
    return {"success": True, "status": "learning_active"}

@app.get("/learning/metrics")
def learning_metrics():
    from pathlib import Path
    import json
    metrics_path = Path(__file__).resolve().parent.parent / "state" / "learning_metrics.json"
    if not metrics_path.exists():
        return {"error": "learning_metrics.json not found"}
    with open(metrics_path) as f:
        data = json.load(f)
    return data

@app.get("/live")
def get_live_data():
    try:
        data = get_best()
        return {"success": True, "data": data}
    except Exception as e:
        return {"success": False, "error": str(e)}
Status: ✅ Verified — /learning/metrics and /learning/status endpoints return valid data.
